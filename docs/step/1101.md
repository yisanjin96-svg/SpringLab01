    SELECT f.amount, COUNT(s.seat_id) AS available_seats
    FROM flight_fare f
    
    LEFT JOIN seat s
        ON s.flight_id = f.flight_id
    AND s.seat_class = f.seat_class
    AND s.status = 'AVAILABLE'
    
    WHERE f.flight_id = 1
        AND f.seat_class = 'BUSINESS'
    
    GROUP BY f.amount;
